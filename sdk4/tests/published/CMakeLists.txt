cmake_minimum_required(VERSION 3.22)
project(apbin_published_package_tests LANGUAGES CXX)
enable_testing()
foreach(required SDK_SOURCE PACKAGE_INPUT_DIR FIXTURE_SOURCE)
  if(NOT DEFINED ${required} OR NOT EXISTS "${${required}}")
    message(FATAL_ERROR "Set ${required} to an existing exact-source directory")
  endif()
endforeach()
file(STRINGS "${SDK_SOURCE}/VERSION" sdk_version LIMIT_COUNT 1)
if(NOT sdk_version STREQUAL "1.0.1")
  message(FATAL_ERROR "Published package audit requires SDK 1.0.1")
endif()
set(PJ_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PJ_BUILD_PORTED_PLUGINS OFF CACHE BOOL "" FORCE)
set(PJ_BUILD_EXPERIMENTAL OFF CACHE BOOL "" FORCE)
set(PJ_INSTALL_SDK OFF CACHE BOOL "" FORCE)
set(PJ_WARNINGS_AS_ERRORS ON CACHE BOOL "" FORCE)
add_subdirectory("${SDK_SOURCE}" sdk)
find_package(nlohmann_json REQUIRED)
if(NOT WIN32)
  message(FATAL_ERROR "This supplementary harness loads the original Windows DLLs")
endif()
foreach(version 1.2.0-pj4.rc.1 1.3.0-pj4.rc.1)
  set(target "published_${version}")
  add_executable(${target} "${PACKAGE_INPUT_DIR}/${version}/source_test.cpp")
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_options(${target} PRIVATE /utf-8)
  target_link_libraries(${target} PRIVATE plotjuggler_sdk::plugin_host nlohmann_json::nlohmann_json)
  add_test(NAME "PublishedSDK101_${version}" COMMAND ${target}
    "${PACKAGE_INPUT_DIR}/${version}/extracted/ardupilot_source_plugin.dll"
    "${FIXTURE_SOURCE}/sdk4/tests" "${FIXTURE_SOURCE}/tests/logs/00000001.BIN")
endforeach()
